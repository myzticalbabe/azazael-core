# Helper script for path translation
# WSL: /mnt/c/Users/clint/Desktop/Azazael
# Windows: C:\Users\clint\Desktop\Azazael

def wsl_to_windows(path: str) -> str:
    """Convert WSL path to Windows path."""
    if path.startswith("/mnt/"):
        drive = path[5].upper()
        windows_path = f"{drive}:{path[6:]}"
        return windows_path.replace("/", "\\")
    return path

def windows_to_wsl(path: str) -> str:
    """Convert Windows path to WSL path."""
    if path[1] == ":":
        drive = path[0].lower()
        wsl_path = f"/mnt/{drive}{path[2:]}"
        return wsl_path.replace("\\", "/")
    return path
