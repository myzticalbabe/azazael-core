import ast, sys
from pathlib import Path

root = Path.home() / "Desktop" / "Azazael_UNIFIED"
dependencies = {}

for py in root.rglob("*.py"):
    if ".venv" in str(py) or "site-packages" in str(py):
        continue
    try:
        content = py.read_text()
        tree = ast.parse(content)
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for alias in node.names:
                    deps = dependencies.get(str(py.relative_to(root)), set())
                    deps.add(alias.name)
                    dependencies[str(py.relative_to(root))] = deps
            elif isinstance(node, ast.ImportFrom):
                if node.module:
                    for alias in node.names:
                        deps = dependencies.get(str(py.relative_to(root)), set())
                        deps.add(f"{node.module}.{alias.name}")
                        dependencies[str(py.relative_to(root))] = deps
    except:
        pass

# Print critical dependencies
print("CRITICAL CROSS-MODULE DEPENDENCIES:")
for file, deps in sorted(dependencies.items()):
    external = [d for d in deps if not d.startswith(".")]
    if external:
        print(f"{file}: {list(external)[:5]}...")

#!/usr/bin/env python3
"""
Automated path detection and modification for migration
"""
import re
from pathlib import Path

class PathDetector:
    def __init__(self, source_root, target_root):
        self.source = Path(source_root)
        self.target = Path(target_root)
        self.patterns = [
            r"C:\\\\Users\\\\clint\\\\Desktop\\\\Azazael_UNIFIED",
            r"/c/Users/clint/Desktop/Azazael_UNIFIED",
            r"~/Desktop/Azazael_UNIFIED",
            r"\$HOME/Desktop/Azazael_UNIFIED",
        ]
        self.files_to_modify = []
    
    def scan(self):
        """Scan all files for paths to update"""
        for py in self.source.rglob("*.py"):
            if ".venv" in str(py) or "site-packages" in str(py):
                continue
            content = py.read_text(encoding="utf-8", errors="ignore")
            matches = []
            for pattern in self.patterns:
                if re.search(pattern, content):
                    matches.append(pattern)
            if matches:
                self.files_to_modify.append({
                    "file": str(py.relative_to(self.source)),
                    "patterns_found": matches
                })
        return self.files_to_modify
    
    def generate_modifications(self):
        """Generate sed-style replacements"""
        modifications = []
        for entry in self.files_to_modify:
            file = entry["file"]
            for pattern in entry["patterns_found"]:
                old = pattern.replace("\\", "\\\\")
                new = str(self.target).replace("\\", "\\\\")
                modifications.append({
                    "file": file,
                    "old_pattern": old,
                    "new_value": new,
                    "command": f'sed -i "s/{old}/{new}/g" {file}'
                })
        return modifications
    
    def create_patch_script(self):
        """Create automated patch script"""
        mods = self.generate_modifications()
        script = '#!/bin/bash\n# AUTO-GENERATED PATCH SCRIPT\n\n'
        for mod in mods:
            script += f'# File: {mod["file"]}\n'
            script += f'sed -i "s/{mod["old_pattern"]}/{mod["new_value"]}/g" {mod["file"]}\n\n'
        
        ps_script = '@echo off\nREM AUTO-GENERATED PATCH SCRIPT\n\n'
        for mod in mods:
            ps_script += f'Replace-FileString -Path "{mod["file"]}" -Find "{mod["old_pattern"]}" -Replace "{mod["new_value"]}"\n'
        
        return script, ps_script

    {"name": "CLI Startup", "cmd": "python azazael_cli.py", "expected": "Memory systems loaded"},

def run_test(test):
    result = subprocess.run(test["cmd"], shell=True, capture_output=True, text=True)
    return {
        "name": test["name"],
        "passed": test["expected"] in result.stdout,
        "stdout": result.stdout[:200],
        "stderr": result.stderr[:200] if result.stderr else ""
    }

def main():
    results = [run_test(t) for t in TESTS]
    passed = sum(1 for r in results if r["passed"])
    print(f"\nMIGRATION TEST RESULTS: {passed}/{len(results)} PASSED\n")
    for r in results:
        status = "[PASS]" if r["passed"] else "[FAIL]"
        print(f"{status} {r['name']}")
        if not r["passed"]:
            print(f"       stdout: {r['stdout'][:100]}")
            print(f"       stderr: {r['stderr'][:100]}")
    return passed == len(results)

if __name__ == "__main__":
    sys.exit(0 if main() else 1)

#!/usr/bin/env python3
"""
Automatically update documentation with new paths
"""
import re
from pathlib import Path

def update_docs():
    root = Path("C:/Azazael")
    
    # Patterns to update
    updates = [
        (r"C:\\\\Users\\\\clint\\\\Desktop\\\\Azazael_UNIFIED", "C:/Azazael"),
        (r"/c/Users/clint/Desktop/Azazael_UNIFIED", "/c/Azazael"),
        (r"~/Desktop/Azazael_UNIFIED", "~/Azazael"),
    ]
    
    for md in root.rglob("*.md"):
        content = md.read_text()
        original = content
        for old, new in updates:
            content = re.sub(old, new, content)
        if content != original:
            md.write_text(content)
            print(f"Updated: {md}")

if __name__ == "__main__":
    update_docs()

#!/usr/bin/env python3
"""
Enhanced rollback with snapshot restore
"""
import subprocess, shutil
from pathlib import Path
import json

class Rollback:
    def __init__(self):
        self.backup_dir = Path("C:/Backup/Azazael")
        self.target = Path("C:/Azazael")
        self.source = Path.home() / "Desktop" / "Azazael_UNIFIED"
    
    def create_backup_snapshot(self):
        """Create timestamped backup before migration"""
        import time
        timestamp = time.strftime("%Y%m%d_%H%M%S")
        snapshot = self.backup_dir / f"pre_migration_{timestamp}"
        snapshot.mkdir(parents=True, exist_ok=True)
        
        # Copy current state
        shutil.copytree(self.source, snapshot / "original", dirs_exist_ok=True)
        
        # Save metadata
        meta = {
            "timestamp": timestamp,
            "source": str(self.source),
            "target": str(self.target)
        }
        (snapshot / "meta.json").write_text(json.dumps(meta, indent=2))
        
        print(f"Backup created: {snapshot}")
        return snapshot
    
    def restore_from_snapshot(self, snapshot_path):
        """Restore from backup snapshot"""
        snapshot = Path(snapshot_path)
        if not snapshot.exists():
            print(f"Snapshot not found: {snapshot_path}")
            return False
        
        # Stop any running processes
        subprocess.run(["taskkill", "/F", "/IM", "python.exe"])
        
        # Remove current target
        if self.target.exists():
            shutil.rmtree(self.target)
        
        # Restore from snapshot
        shutil.copytree(snapshot / "original", self.source, dirs_exist_ok=True)
        
        print(f"Restored from: {snapshot}")
        return True
    
    def quick_rollback(self):
        """Quick rollback - restore original directory"""
        if not self.source.exists():
            print("Original directory not found")
            return False
        
        # Stop processes
        subprocess.run(["taskkill", "/F", "/IM", "python.exe"])
        
        # Remove target
        if self.target.exists():

        print("Quick rollback complete")
        return True
